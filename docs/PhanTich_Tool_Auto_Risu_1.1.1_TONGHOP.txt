# 4. RisuGame.dll — LOGIC HOOK MẠNG (DECOMPILE ĐẦY ĐỦ)
########################################################################
DLL inject vào flash.exe. 2 hàm cốt lõi:

4.1 DllMain (cài hook) — pseudocode đã chú giải:
  BOOL DllMain(hMod, reason){
    if (reason==DLL_PROCESS_ATTACH){
      DisableThreadLibraryCalls(hMod);
      if (ShouldActivate()==0){
        ws2 = LoadLibraryA("ws2_32.dll");
        realConnect = GetProcAddress(ws2,"connect");
        InstallInlineHook(realConnect, &MyConnect, &g_realConnect); // lưu connect() gốc
        FlushInstructionCache(realConnect);
        // ép luôn HTTP/WinINet qua proxy nội bộ:
        opt = {scheme=3, proxy="127.0.0.1:45678"};
        UrlMkSetSessionOption(INTERNET_OPTION_PROXY/*0x26*/,&opt,...);
        InternetSetOptionA(0, 0x26, &opt, ...);
      }
    }
    return TRUE;
  }

4.2 connect() GIẢ đã hook — bẻ hướng qua SOCKS5 (RFC1928, NO-AUTH):
  int MyConnect(SOCKET s, sockaddr_in* addr, int len){
    if (addr->sin_family==AF_INET){
      destIp = inet_ntoa(addr->sin_addr); destPort = ntohs(addr->sin_port);
      Log("[Connect Hook] App dang co ket noi den -> %s:%d");
      socksPort = atoi(getenv("RISU_SOCKS_PORT"));
      if (strcmp(destIp,"182.92.148.45")==0 && socksPort>0){     // chỉ chặn IP game
        Log("[Proxy] Bat duoc ket noi den Game Server ... chuyen huong qua SOCKS5...");
        ioctlsocket(s, FIONBIO, &(u_long)0);                     // về blocking
        g_realConnect(s, {127.0.0.1:socksPort}, 16);             // nối SOCKS5 nội bộ
        send(s, {05,01,00}, 3);  recv(s, resp, 2);               // greeting -> mong {05,00}
        send(s, {05,01,00,01, IP(4), PORT(2)}, 10);              // CONNECT tới IP game gốc
        recv(s, rep, 10);
        if (rep[0]==05 && rep[1]==00){ ioctlsocket(s,FIONBIO,&(u_long)1); return 0; }
        return -1;
      }
    }
    return g_realConnect(s, addr, len);                          // IP khác -> đi thẳng
  }
